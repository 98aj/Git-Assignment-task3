task 3:

1. mkdir task3
2. git init
3. touch main.txt
4. git add main.txt
5. git commit -m 'Inital commit of adding main file on master branch'
6. git remote add origin https://[secret-key-token]@github.com/98aj/Git-Assignment-task3.git
7. git push -u origin master
8. git branch develop (creating branch form master)
9. git branch f1 (creating branch form master)
10. git branch f2 (creating branch form master)
11. git checkout develop
12. touch develop.txt
13. git add develop.txt
14. git commit -m 'Commiting develop file to develop branch'
15. git push -u origin develop
16. git switch f1
17. touch f1.txt
18. git add f1.txt
19. git commit -m 'Commiting f1 file to f1 branch'
20. git push -u origin f1
21. git switch f2
21. touch f2.txt
23. git add f2.txt
22. git commit -m 'Commiting f2 file to f2 branch'
23. git push -u origin f2
24. git switch master
25. git branch -d f2 (delete f2 branch locally)
26. git push origin --delete f2 (delete branch on github from CLI)
